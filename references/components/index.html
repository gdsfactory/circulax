    def component(self, name: str = "Junction") -> type[CircuitComponent]:
    """Return a two-terminal component with ports ``anode`` and ``cathode``.

    Its runtime parameters are ``length_um`` plus :attr:`defaults`; the
    component carries ``length * current(v)`` and stores ``length * charge(v)``.
    """
    junction = self

    def physics(signals: Signals, **kw: Any) -> PhysicsReturn:
        v = jnp.real(signals.anode - signals.cathode)
        length_m = kw["length_um"] * 1e-6
        current = length_m * junction.current_per_m(v, kw)
        charge = length_m * junction.charge_per_m(v, kw)
        return {"anode": current, "cathode": -current}, {"anode": charge, "cathode": -charge}

    _set_signature(physics, name, {"length_um": _RUNTIME_DEFAULTS["length_um"]}, self.parameters)
    return component(ports=("anode", "cathode"))(physics)

    def netlist(self, *, length: float, n_sections: int = 1) -> dict:  # noqa: C901, PLR0915 -- local RLC branch builders
    """Build an unterminated symmetric-T ladder from R, L and C primitives.

    Ports ``p1`` and ``p2`` are single-ended relative to the internal common
    ground. ``tap_0`` ... ``tap_{n_sections-1}`` expose section midpoints for
    junction loading. No source, termination, optical model or delay is added.
    Increasing ``n_sections`` reduces the spatial discretization error.
    """
    if not np.isfinite(length) or length <= 0:
        msg = "length must be finite and positive, in metres"
        raise ValueError(msg)
    if isinstance(n_sections, bool) or not isinstance(n_sections, Integral) or n_sections < 1:
        msg = "n_sections must be a positive integer"
        raise ValueError(msg)
    instances = {"GND": {"component": "ground"}}
    connections: dict[str, str] = {}
    ports = {}
    serial = 0

    def put(kind: str, value: float, start: str | None, end: str | None = None) -> tuple[str, str]:
        nonlocal serial
        serial += 1
        name = f"{kind}{serial}"
        instances[name] = {"component": kind, "settings": {kind: float(value)}}
        p1, p2 = f"{name},p1", f"{name},p2"
        if start is not None:
            connections[p1] = start
        if end is not None:
            connections[p2] = end
        return p1, p2

    def series(start: str | None, distance: float) -> str:
        tip = start
        first = None
        if self.series.constant > 0:
            first, tip = put("R", self.series.constant * distance, tip)
        if self.series.linear > 0:
            begin, tip = put("L", self.series.linear * distance, tip)
            first = first or begin
        for w, p in zip(self.series.weights, self.series.poles, strict=True):
            begin, end = put("R", w * distance, tip)
            put("L", w * distance / p, begin, end)
            first, tip = first or begin, end
        if tip is None:
            msg = "The line must have nonzero series impedance"
            raise ValueError(msg)
        if "p1" not in ports:
            ports["p1"] = first
        return tip

    def shunt(node: str, distance: float) -> None:
        if self.shunt.constant > 0:
            put("R", 1 / (self.shunt.constant * distance), node, "GND,p1")
        if self.shunt.linear > 0:
            put("C", self.shunt.linear * distance, node, "GND,p1")
        for w, p in zip(self.shunt.weights, self.shunt.poles, strict=True):
            _, tip = put("R", 1 / (w * distance), node)
            put("C", w * distance / p, tip, "GND,p1")

    last = None
    d = length / n_sections
    for k in range(n_sections):
        midpoint = series(last, d / 2)
        shunt(midpoint, d)
        ports[f"tap_{k}"] = midpoint
        last = series(midpoint, d / 2)
    ports["p2"] = last
    return {"instances": instances, "connections": connections, "ports": ports}

    @component(ports=("inp", "out"), states=("i_out", "x", "z"), holomorphic=True)
def Bessel3(signals: Signals, f3db: float = 35e9) -> PhysicsReturn:
    """Unity-DC-gain third-order Bessel low-pass, normalized at -3 dB.

    The delay-normalized prototype is ``15/(p**3+6*p**2+15*p+15)``.
    Its -3 dB angular frequency is 1.7556723686812106; scaling by it makes
    ``f3db`` the voltage magnitude cutoff. The DC group delay is
    ``1.7556723686812106/(2*pi*f3db)``. This buffered block includes that
    filter delay; add physical flight time separately when needed.
    """
    r3db = 1.7556723686812106
    w = 2.0 * jnp.pi * f3db / r3db
    return {
        "inp": 0.0 * signals.inp,
        "out": signals.i_out,
        "i_out": w * (15 * signals.out + 15 * signals.x + 6 * signals.z - 15 * signals.inp),
        "x": -w * signals.x,
        "z": -w * signals.z,
    }, {"i_out": signals.z, "x": signals.out, "z": signals.x}

    @component(ports=("inp", "out"), states=("i_out", "x"), holomorphic=True)
def Biquad(signals: Signals, f3db: float = 10e9, Q: float = 0.5773502692) -> PhysicsReturn:
    """Unity-DC-gain second-order low-pass with -3 dB frequency and quality factor.

    ``H(s)=w0**2/(s**2+w0*s/Q+w0**2)``. ``w0`` is computed so the magnitude
    is 1/sqrt(2) at ``f3db`` for every positive Q. For Q>1/sqrt(2), the
    response has peaking. This is an ideal buffered voltage block.
    """
    b = 1.0 / Q**2 - 2.0
    # Positive root of u**2 + b*u - 1 = 0, without cancellation at small Q.
    u = 2.0 / (b + jnp.sqrt(b**2 + 4.0))
    w0 = 2.0 * jnp.pi * f3db / jnp.sqrt(u)
    return {
        "inp": 0.0 * signals.inp,
        "out": signals.i_out,
        "i_out": w0 * (signals.out + signals.x / Q - signals.inp),
        "x": -w0 * signals.x,
    }, {"i_out": signals.x, "x": signals.out}

        {"i1": -(L1 * signals.i1 + m * signals.i2), "i2": -(m * signals.i1 + L2 * signals.i2)},
    )

    OpticalDelayLine ¤

    OpticalDelayLine(
    signals: Signals,
    length_um: float = 100.0,
    loss_dB_cm: float = 1.0,
    neff: float = 2.4,
    n_group: float = 4.0,
    wavelength_nm: float = 1310.0,
) -> PhysicsReturn

    Waveguide modelled as an explicit time-of-flight delay line (real group delay).

    Unlike :func:OpticalWaveguide -- which is a steady-state S-matrix stamp only valid for CW/frequency-domain analysis -- this component enforces the actual transient envelope delay: the output field at p2 is the input field at p1 from tau = length_um * n_group / c seconds ago, attenuated and phase-shifted, via a VCVS-style constraint (same pattern as :func:TunableBeamSplitter). p1 carries no self-stamp and must be driven by a connected source/component, matching that same convention.

    def tabulated(voltages: ArrayLike, values: ArrayLike) -> VoltageFunction:
    """Return a piecewise-linear function of voltage from a sampled table.

    Use it for ``delta_neff``, ``loss_dB_cm`` or a junction ``current`` when
    data is tabulated. Values are held constant outside the table. For a measured
    capacitance table use :func:`charge_from_capacitance_table` instead, which
    integrates exactly and keeps the capacitance continuous.
    """
    v, y = _table(voltages, values, "tabulated")
    v_j, y_j = jnp.asarray(v), jnp.asarray(y)

    def table(voltage: Any) -> Any:
        return jnp.interp(voltage, v_j, y_j)

    return table
